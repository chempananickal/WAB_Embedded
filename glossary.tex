\newglossaryentry{iot}{
    name={IoT},
    description={Internet of Things}
}
\newglossaryentry{logp}{
    name={logP},
    description={Logarithm of the octanol-water partition coefficient. 
    A measure of a compound's hydrophobicity, or how well it dissolves in fats versus water}
}
\newglossaryentry{tflite}{
    name={TFLite},
    description={TensorFlow Lite, a framework for deploying machine learning models on mobile and embedded devices}
}
\newglossaryentry{esp32}{
    name={ESP32},
    description={Espressif Systems' low-cost, low-power system on a chip (SoC) with integrated Wi-Fi and Bluetooth capabilities}
}
\newglossaryentry{ZINC20}{
    name={ZINC20},
    description={ZINC Is Not Commercial (2020), a free database of commercially available compounds for virtual screening}
}
\newglossaryentry{rf}{
    name={RF},
    description={Random Forest, an ensemble learning method for classification and regression tasks that constructs multiple decision trees during training}
}
\newglossaryentry{svm}{
    name={SVM},
    description={Support Vector Machine, a supervised machine learning algorithm used for classification and regression tasks that finds the hyperplane that best separates the classes in the feature space}
}
\newglossaryentry{smiles}{
    name={SMILES},
    description={Simplified Molecular Input Line Entry System, a notation that allows a user to represent a chemical structure as a string of ASCII characters}
}
\newglossaryentry{fingerprint}{
    name={fingerprint},
    description={A binary vector representation of a molecule, where each bit represents the presence or absence of a particular substructure or feature in the molecule}
}
\newglossaryentry{qrcode}{
    name={QR code},
    description={Quick Response Code, a type of matrix barcode (or two-dimensional barcode) that can store information such as a URL, text, or other data}
}
\newglossaryentry{lipophilicity}{
    name={lipophilicity},
    description={The chemical property of being fat-soluble, often quantified by the octanol-water partition coefficient (logP)}
}

\newglossaryentry{relu}{
    name={ReLU},
    description={Rectified Linear Unit, a type of activation function used in neural networks that outputs the input directly if it is positive, otherwise, it outputs zero}
}
