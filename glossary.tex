% Glossary entries
\newglossaryentry{iot}{
    name={IoT},
    description={Internet of Things}
}
\newglossaryentry{logp}{
    name={LogP},
    text={logP},
    description={Logarithm of the octanol-water partition coefficient. 
    A measure of a compound's hydrophobicity, or how well it dissolves in fats versus water}
}
\newglossaryentry{tflite}{
    name={TensorFlow Lite (TFLite)},
    text={TFLite},
    description={A framework for deploying machine learning models on mobile and embedded devices. Currently being rebranded as LiteRT}
}
\newglossaryentry{esp32}{
    name={ESP32},
    description={Espressif Systems' low-cost, low-power system on a chip (SoC) with integrated Wi-Fi and Bluetooth capabilities}
}
\newglossaryentry{ZINC20}{
    name={ZINC20},
    description={ZINC Is Not Commercial (2020), a free database of commercially available compounds for virtual screening}
}
\newglossaryentry{rf}{
    name={Random Forest (RF)},
    text={RF},
    description={An ensemble learning method for classification and regression tasks that constructs multiple bootstrapped decision trees during training}
}
\newglossaryentry{svm}{
    name={Support Vector Machine (SVM)},
    text={SVM},
    description={A supervised machine learning algorithm used for classification and regression tasks that finds the hyperplane that best separates the classes in the feature space}
}
\newglossaryentry{smiles}{
    name={Simplified Molecular Input Line Entry System (SMILES)},
    text={SMILES},
    description={A notation that allows a user to represent a chemical structure as a string of ASCII characters}
}
\newglossaryentry{fingerprint}{
    name={Fingerprint},
    text={fingerprint},
    description={An algorithmically constructed binary vector representation of a molecule, where each bit represents the presence or absence of a particular substructure or feature in the molecule}
}
\newglossaryentry{qrcode}{
    name={QR code},
    description={Quick Response Code, a type of matrix barcode (or two-dimensional barcode) that can store information such as a URL, text, or other data}
}
\newglossaryentry{lipophilicity}{
    name={Lipophilicity},
    text={lipophilicity},
    description={The chemical property of being fat-soluble, characterized by a highly positive logP value}
}

\newglossaryentry{bioavailability}{
    name={Bioavailability},
    text={bioavailability},
    description={The proportion of a drug or other substance that enters the circulation when introduced into the body and is able to have an active effect}
}

\newglossaryentry{hydrophilicity}{
    name={Hydrophilicity},
    text={hydrophilicity},
    description={The chemical property of being water-soluble, characterized by a highly negative logP value}
}

\newglossaryentry{relu}{
    name={Rectified Linear Unit (ReLU)},
    text={ReLU},
    description={A type of activation function used in neural networks that outputs the input directly if it is positive, otherwise, it outputs zero}
}

\newglossaryentry{xgboost}{
    name={Extreme Gradient Boosting (XGBoost)},
    text={XGBoost},
    description={An ensemble learning method that uses gradient boosting to combine the predictions of multiple weak learners (usually decision trees) to create a stronger predictive model}
}

\newglossaryentry{adam}{
    name={Adaptive Moment Estimation (Adam)},
    text={Adam},
    description={An optimization algorithm used in training machine learning models that computes adaptive learning rates for each parameter}
}

\newglossaryentry{mlp}{
    name={Multi-layer Perceptron (MLP)},
    text={MLP},
    description={A feed-forward artificial neural network composed of an input layer, one or more fully connected hidden layers, and an output layer}
}
